### 用户反馈核心诉求

1. **机构识别时，"公司/有限公司/电子厂"等标记为 POI 是错误的，应该标为 ORG。** POI 识别时怎么标不管，但机构名识别时就该标 ORG。
2. **"电子厂"在 SQL 切分时应处理成 d3(电子)+d4(厂)**。因无法区分"黄河路店"和"玩具店"，被直接归入 d4，对后续有污染。**v2.2 应直接修复高频 d4 污染。**
3. **2 段及以上序列，如果类组合的词频 <10，在亿级数据中肯定有问题。** 应提取为 badcase 测试集，后续评估。
4. **查表应该是尺子——用来调 score——而不是直接替代 argmax。** 但规则 2（d4∉SQL CSV→POI）保留直接 return，有数据验证支撑。

```python
#!/usr/bin/env python3
"""统计 org d4 中 d3+d4 可拆分后缀 — 只读，不改数据"""
import os, sys, pickle

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

DICT_DIR = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "dicts")
SPLIT_SUFFIXES = {"厂", "店", "行", "部", "坊", "室", "所", "站", "场"}


def main():
    with open(os.path.join(DICT_DIR, "org_v1.2.1.pkl"), "rb") as f:
        org = pickle.load(f)
    d4_freqs = org["d4_freqs"]
    d3_freqs = org["d3_freqs"]

    polluted = []
    for d4_word, d4_freq in d4_freqs.items():
        if len(d4_word) < 3:
            continue
        for pos in range(len(d4_word) - 1, 0, -1):
            prefix = d4_word[:pos]
            suffix = d4_word[pos:]
            if suffix in SPLIT_SUFFIXES and prefix in d3_freqs:
                polluted.append((d4_word, d4_freq, prefix, d3_freqs[prefix], suffix))
                break

    print(f"d4 后缀总数: {len(d4_freqs):,}")
    print(f"d3 中可拆分: {len(polluted):,} ({100*len(polluted)/len(d4_freqs):.1f}%)")
    print()

    for threshold in [100, 500, 1000, 5000]:
        count = sum(1 for p in polluted if p[3] >= threshold)
        print(f"  d3 freq >= {threshold:>5}: {count:>5} d4 entries ({100*count/len(d4_freqs):.1f}%)")

    print(f"\nTop 30 (d3 freq >= 1000):")
    heavy = [p for p in polluted if p[3] >= 1000]
    heavy.sort(key=lambda x: -x[3])
    for d4w, d4f, prefix, d3f, sfx in heavy[:30]:
        print(f"  {d4w:12s}  d4_freq={d4f:>8,}  → d3({prefix}:{d3f:,}) + d4({sfx})")


if __name__ == "__main__":
    main()
```

## 六、阶段 C — 词典与规则优化

### C1. 新增 split_polluted_d4（拆"电子厂"→d3"电子"+d4"厂"）

**用户原话**："电子厂应处理成 d3电子+d4厂"。v2.2 直接修复高频 d4 污染。

**修改文件**：`tools/org_aggregate.py`

新增函数（在 `main()` 中 `prune()` 之后、`merge_gov_seed()` 之前调用）：

```python
SPLIT_SUFFIXES = {"厂", "店", "行", "部", "坊", "室", "所", "站", "场"}
D3_FREQ_THRESHOLD = 5000


def split_polluted_d4(pruned, merged):
    """拆分 d4 中可分解为 d3+单字后缀的条目。
    例: 电子厂(34,926) → d3 新增 电子(8,765) + d4 新增 厂
    条件: len(d4)>=3, 末尾单字∈SPLIT_SUFFIXES, 前缀在 merged d3 中 freq>=5000
    """
    d3 = dict(pruned["d3_freqs"])
    d4 = dict(pruned["d4_freqs"])
    merged_d3 = merged.get("d3_freqs", {})

    split_count = 0
    split_details = []

    for d4_word in list(d4.keys()):
        if len(d4_word) < 3:
            continue
        suffix = d4_word[-1]
        if suffix not in SPLIT_SUFFIXES:
            continue
        prefix = d4_word[:-1]
        if len(prefix) < 2:
            continue
        prefix_freq = merged_d3.get(prefix, 0)
        if prefix_freq < D3_FREQ_THRESHOLD:
            continue

        d4_freq = d4.pop(d4_word)
        if prefix not in d3:
            d3[prefix] = prefix_freq
        if suffix not in d4:
            d4[suffix] = max(d4.get(suffix, 0), d4_freq)
        split_count += 1
        split_details.append((d4_word, d4_freq, prefix, prefix_freq, suffix))

    pruned["d3_freqs"] = d3
    pruned["d4_freqs"] = d4

    print(f"D4 pollution split: {split_count} entries split", flush=True)
    for d4w, d4f, pfx, pfx_f, sfx in split_details[:10]:
        print(f"  {d4w} (d4={d4f:,}) → d3({pfx}:{pfx_f:,}) + d4({sfx})", flush=True)

    return pruned, split_count
```

**调用点**（`org_aggregate.py main()` 中）：

```python
pruned = prune(merged, quality)
pruned, split_n = split_polluted_d4(pruned, merged)
pruned, gov_injected, gov_boosted = merge_gov_seed(pruned)
```

**设计要点**：

| 条件 | 理由 |
|------|------|
| SPLIT_SUFFIXES 只含单字 | 只有'厂/店/行/部'等单字实体后缀才拆分，'公司''有限公司'不拆 |
| D3_FREQ_THRESHOLD=5000 | 确保"电子""玩具"等真行业词被拆，冷僻组合不拆 |
| "黄河路店"不会被拆 | "黄河路"不在 d3 中（它是 d1 道路名） |
| "玩具店"会被拆 | "玩具"在 d3 中 freq≥5000，"店"∈SPLIT_SUFFIXES |
| 拆分后用 max 合并 | 如果"厂"已在 d4 中，取较大 freq |

**原则**：规则 2（d4∉SQL CSV→POI）保留直接 return（数据验证过：84% POI→UNKNOWN、81% POI→ADDR 属于此类）。其余查表命中改为调 score：

| suffix_table=POI_STORE + has_geo | `return "POI"` | `scores["POI"] += 0.20` |

**新 `_decide` 实现**：

```python
def _decide(self, f, scores):
    scores = dict(scores)
    from core.csv_index import CsvIndex
    idx = CsvIndex.instance()

    if (f["has_comp"] or f["has_gov"]) and f["org_has_poi_terminal"]:
        return "ORG", scores["ORG"]

    d4 = f.get("d4_suffix")
    if d4:
        if d4 not in self.sql_suffixes:
            return "POI", max(scores["POI"], 0.70)

        cat = self._lookup_d4_category(d4)
        if cat == "POI_FAC":
            scores["POI"] = min(scores["POI"] + 0.30, 0.95)
        elif cat == "POI_STORE":
            if f["has_geo"] or self._double_negative(d4, idx):
                scores["POI"] = min(scores["POI"] + 0.25, 0.95)
            else:
                scores["POI"] = min(scores["POI"] + 0.15, 0.90)
        elif cat == "ORG":
            scores["ORG"] = min(max(scores["ORG"], 0.70), 0.98)

        if cat in (None, "AMBIGUOUS", "UNKNOWN"):
            cat_sfx = self._lookup_suffix_category(d4)
            if cat_sfx == "POI_FAC":
                scores["POI"] = min(scores["POI"] + 0.30, 0.95)
            elif cat_sfx == "POI_STORE":
                if f["has_geo"] or self._double_negative(d4, idx):
                    scores["POI"] = min(scores["POI"] + 0.20, 0.95)
                else:
                    scores["POI"] = min(scores["POI"] + 0.10, 0.90)
            elif cat_sfx == "ORG":
                scores["ORG"] = min(max(scores["ORG"], 0.65), 0.98)

    best_cat = max(scores, key=scores.get)
    best_val = scores[best_cat]
    if best_val < self.T_UNKNOWN:
        return "UNKNOWN", best_val
    return best_cat, best_val
```

**预期效果**：POI recall 从 0.52→0.55~0.58，不降低 ORG recall。完成后重新标定 T_UNKNOWN（扫描 0.05~0.30）。

### C4. LOC 道路名相邻合并拆分

**修改文件**：`tools/merge_gaode_roads.py`

**问题**："未来路货栈街"被 DP 合成一段。两个有效道路名同时存在导致 Trie 长匹配偏好走连片路径。

**修复**：在 `main()` 中道路名合并完成后增加后处理，直接 `del` 可拆分的组合条目：

```python
def _split_adjacent_roads(road_dict):
    road_endings = ('路', '街', '道', '巷', '弄', '胡同', '大街', '大道', '环路')
    roads = {w for w in road_dict if w.endswith(road_endings)}

    deleted = 0
    for w in list(road_dict.keys()):
        if len(w) < 5 or not w.endswith(road_endings):
            continue
        for pos in range(2, len(w) - 1):
            left, right = w[:pos], w[pos:]
            if left in roads and right in roads:
                del road_dict[w]
                deleted += 1
                break

    return deleted
```

**调用**：

```python
deleted = _split_adjacent_roads(road_dict)
print(f"Split adjacent roads: {deleted} merged entries deleted")
```
