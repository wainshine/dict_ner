### 12. 消除 brand/model.py 训练代码重复

```python
def _count_ngrams(self, text: str, weight: int = 1):
    """提取 bigram + unigram 计数"""
    for j in range(len(text)):
        self.unigram_counts[text[j]] += weight
    for j in range(len(text) - 1):
        self.bigram_counts[text[j] + text[j + 1]] += weight

def _bigram_log_prob(self, text: str) -> float:
    """共享的 bigram log-prob 计算"""
    if len(text) < 2:
        return None
    total_log = 0.0
    count = 0
    for i in range(len(text) - 1):
        prob = self.bigram_prob(text[i], text[i + 1])
        if prob > 1e-12:
            total_log += math.log(prob)
            count += 1
    return total_log, count
```
